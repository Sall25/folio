export const PROPERTY_TYPE_MAP = new Map(
  PROPERTY_TYPE_META.map((m) => [m.type, m]),
);

export function getTypeMeta(type: PropertyType): PropertyTypeMeta {
  return PROPERTY_TYPE_MAP.get(type) ?? PROPERTY_TYPE_META[0];
}

// Types that can be changed to — title can never be changed
export const CHANGEABLE_TYPES = PROPERTY_TYPE_META.filter(
  (m) => m.type !== "title",
);
