import { NodeViewWrapper, type NodeViewProps } from "@tiptap/react";
import { useRef } from "react";

import type { DatabaseAttrs } from "./types";
import { useDatabase } from "./use-database";
import { DatabaseToolbar } from "./database-toolbar";
import { DatabaseTableView } from "./database-table-view";

import "./database.scss";

export function DatabaseNodeView({ node, editor }: NodeViewProps) {
  const attrs = node.attrs as DatabaseAttrs;
  const containerRef = useRef<HTMLDivElement>(null);
  const db = useDatabase(attrs, editor);

  const panelOpen = db.openPropertyId !== null;

  return (
    <NodeViewWrapper>
      <div className="db-root" ref={containerRef}>
        {/* ── Toolbar ── */}
        <DatabaseToolbar db={db} />

        {/* ── Body: table + optional side panel ── */}
        <div className="db-body">
          {/* Table shrinks when panel is open */}
          <div
            className={`db-table-wrap ${panelOpen ? "db-table-wrap--shrunk" : ""}`}
          >
            {db.activeView?.type === "table" && <DatabaseTableView db={db} />}
            {db.activeView?.type === "board" && (
              <div className="db-placeholder">Board view coming soon</div>
            )}
            {db.activeView?.type === "list" && (
              <div className="db-placeholder">List view coming soon</div>
            )}
            {db.activeView?.type === "gallery" && (
              <div className="db-placeholder">Gallery view coming soon</div>
            )}
          </div>
        </div>
      </div>
    </NodeViewWrapper>
  );
}
