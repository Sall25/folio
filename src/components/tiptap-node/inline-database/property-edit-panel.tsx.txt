export interface PropertyEditPanelProps {
  property: DatabaseProperty;
  /** All properties in the database — needed by FormulaProperty for prop references */
  allProperties: DatabaseProperty[];
  onUpdate: (patch: Partial<Omit<DatabaseProperty, "id">>) => void;
  onDelete: () => void;
  onDuplicate: () => void;
  onClose: () => void;
  /** Panel navigation stack — owned by useDatabase */
  panelStack: PanelView[];
  pushPanel: (view: PanelView) => void;
  popPanel: () => void;
  resetPanel: () => void;
  isFrozen: boolean;
  isHidden: boolean;
  isUnwrapped: boolean;
  onFreeze: () => void;
  onHide: () => void;
  onUnwrap: () => void;
}

// ─────────────────────────────────────────────────────────────────────────────
// Header — shared across all panel views
// ─────────────────────────────────────────────────────────────────────────────

interface PanelHeaderProps {
  title: string;
  canGoBack: boolean;
  onBack: () => void;
}

function PanelHeader({ title, canGoBack, onBack }: PanelHeaderProps) {
  return (
    <div
      style={{
        display: "flex",
        alignItems: "center",
        padding: "6px 4px 4px",
        gap: 4,
        borderBottom: "0.5px solid var(--tt-border-color)",
        marginBottom: 4,
      }}
    >
      {canGoBack && (
        <Button
          variant="ghost"
          style={{ width: 24, height: 24, padding: 0, flexShrink: 0 }}
          onClick={onBack}
          aria-label="Back"
        >
          <ChevronLeft
            style={{ width: 14, height: 14 }}
            className="tiptap-button-icon"
          />
        </Button>
      )}
      <span
        style={{
          fontSize: 11,
          fontWeight: 500,
          textTransform: "uppercase",
          letterSpacing: "0.07em",
          color: "var(--tt-gray-light-a-500)",
          flex: 1,
        }}
      >
        {title}
      </span>
    </div>
  );
}

// ─────────────────────────────────────────────────────────────────────────────
// Type selector row — shown in main panel when type is not title
// ─────────────────────────────────────────────────────────────────────────────

interface TypeSelectorProps {
  currentType: PropertyType;
  onSelect: (type: PropertyType) => void;
  showList: boolean;
  onToggleList: () => void;
}

function TypeSelector({
  currentType,
  onSelect,
  showList,
  onToggleList,
}: TypeSelectorProps) {
  const meta = getTypeMeta(currentType);
  const Icon = (LucideIcons as any)[meta.icon];

  return (
    <CardItemGroup>
      <CardGroupLabel>Type</CardGroupLabel>

      <Button
        variant="ghost"
        style={{
          width: "100%",
          height: 30,
          justifyContent: "flex-start",
          gap: 8,
        }}
        onClick={onToggleList}
      >
        {Icon && (
          <Icon
            style={{ width: 14, height: 14 }}
            className="tiptap-button-icon"
          />
        )}
        <span style={{ fontSize: 13, flex: 1, textAlign: "left" }}>
          {meta.label}
        </span>
        <ChevronRight
          style={{
            width: 13,
            height: 13,
            transform: showList ? "rotate(90deg)" : "none",
            transition: "transform 0.15s",
          }}
          className="tiptap-button-icon-sub"
        />
      </Button>

      {showList && (
        <div style={{ paddingLeft: 4 }}>
          {CHANGEABLE_TYPES.map((t) => {
            const TIcon = (LucideIcons as any)[t.icon];
            return (
              <Button
                key={t.type}
                variant="ghost"
                style={{
                  width: "100%",
                  height: 28,
                  justifyContent: "flex-start",
                  gap: 8,
                }}
                onClick={() => onSelect(t.type)}
              >
                <span
                  style={{
                    width: 14,
                    display: "flex",
                    alignItems: "center",
                    justifyContent: "center",
                  }}
                >
                  {currentType === t.type && (
                    <span
                      style={{
                        width: 6,
                        height: 6,
                        borderRadius: "50%",
                        background: "var(--tt-brand-color-500)",
                      }}
                    />
                  )}
                </span>
                {TIcon && (
                  <TIcon
                    style={{ width: 13, height: 13 }}
                    className="tiptap-button-icon"
                  />
                )}
                <span style={{ fontSize: 13 }}>{t.label}</span>
              </Button>
            );
          })}
        </div>
      )}
    </CardItemGroup>
  );
}

// ─────────────────────────────────────────────────────────────────────────────
// Main panel view
// ─────────────────────────────────────────────────────────────────────────────

interface MainPanelProps {
  property: DatabaseProperty;
  allProperties: DatabaseProperty[];
  onUpdate: (patch: Partial<Omit<DatabaseProperty, "id">>) => void;
  onDelete: () => void;
  onDuplicate: () => void;
  pushPanel: (view: PanelView) => void;
  isTitle: boolean;
  onFreeze: () => void;
  isFrozen: boolean;
  onUnwrap: () => void;
  isUnwrapped: boolean;
  onHide: () => void;
}

function MainPanel({
  property,
  /*allProperties,*/
  onUpdate,
  onDelete,
  onDuplicate,
  pushPanel,
  isTitle,
  onFreeze,
  isFrozen,
  onUnwrap,
  isUnwrapped,
  onHide,
}: MainPanelProps) {
  const [showTypeList, setShowTypeList] = useState(false);
  const nameRef = useRef<HTMLInputElement>(null);

  useEffect(() => {
    nameRef.current?.focus();
    nameRef.current?.select();
  }, []);

  const handleTypeChange = (type: PropertyType) => {
    const config = (DEFAULT_CONFIGS as Record<PropertyType, PropertyConfig>)[
      type
    ];
    onUpdate({ config });
    setShowTypeList(false);
  };

  const updateConfig = (config: PropertyConfig) => onUpdate({ config });

  const { config } = property;

  return (
    <CardBody style={{ minWidth: 260 }}>
      {/* ── Name ── */}
      <CardItemGroup>
        <input
          ref={nameRef}
          style={{
            width: "100%",
            border: "0.5px solid var(--tt-border-color)",
            borderRadius: "var(--tt-radius-md)",
            padding: "5px 8px",
            fontSize: 13,
            fontFamily: "inherit",
            background: "var(--tt-bg-color)",
            color: "var(--tt-gray-light-a-900)",
            outline: "none",
          }}
          value={property.name}
          onChange={(e) => onUpdate({ name: e.target.value })}
          placeholder="Property name"
          onFocus={(e) =>
            (e.target.style.borderColor = "var(--tt-brand-color-400)")
          }
          onBlur={(e) =>
            (e.target.style.borderColor = "var(--tt-border-color)")
          }
        />
      </CardItemGroup>

      {/* ── Type selector — hidden for title ── */}
      {!isTitle && (
        <TypeSelector
          currentType={config.type}
          onSelect={handleTypeChange}
          showList={showTypeList}
          onToggleList={() => setShowTypeList((v) => !v)}
        />
      )}

      {/* ── Type-specific sections ── */}

      {(config.type === "select" || config.type === "multi_select") && (
        <SelectOptionsSection config={config} onChange={updateConfig} />
      )}

      {config.type === "number" && (
        <NumberFormatSection config={config} onChange={updateConfig} />
      )}

      {config.type === "date" && (
        <DateFormatSection config={config} onChange={updateConfig} />
      )}

      {/* Status — drill into StatusEditDisplay */}
      {config.type === "status" && (
        <CardItemGroup>
          <CardGroupLabel>Options</CardGroupLabel>
          <Button
            variant="ghost"
            style={{
              width: "100%",
              height: 30,
              justifyContent: "flex-start",
              gap: 8,
            }}
            onClick={() => pushPanel({ type: "status-editor" })}
          >
            <span style={{ fontSize: 13, flex: 1, textAlign: "left" }}>
              {config.groups.reduce(
                (acc: any, g: { items: string | any[] }) =>
                  acc + g.items.length,
                0,
              )}{" "}
              options
            </span>
            <ChevronRight
              style={{ width: 13, height: 13 }}
              className="tiptap-button-icon-sub"
            />
          </Button>
        </CardItemGroup>
      )}

      {/* Person — drill into PersonProperty */}
      {config.type === "person" && (
        <CardItemGroup>
          <CardGroupLabel>Settings</CardGroupLabel>
          <Button
            variant="ghost"
            style={{
              width: "100%",
              height: 30,
              justifyContent: "flex-start",
              gap: 8,
            }}
            onClick={() => pushPanel({ type: "person-limit" })}
          >
            <span style={{ fontSize: 13, flex: 1 }}>Limit</span>
            <span style={{ fontSize: 12, color: "var(--tt-gray-light-a-400)" }}>
              {config.limit === "no-limit" ? "No limit" : "Single person"}
            </span>
            <ChevronRight
              style={{ width: 13, height: 13 }}
              className="tiptap-button-icon-sub"
            />
          </Button>
          <Button
            variant="ghost"
            style={{
              width: "100%",
              height: 30,
              justifyContent: "flex-start",
              gap: 8,
            }}
            onClick={() => pushPanel({ type: "person-default" })}
          >
            <span style={{ fontSize: 13, flex: 1 }}>Default</span>
            <span style={{ fontSize: 12, color: "var(--tt-gray-light-a-400)" }}>
              {config.default === "no-default" ? "No default" : "Me"}
            </span>
            <ChevronRight
              style={{ width: 13, height: 13 }}
              className="tiptap-button-icon-sub"
            />
          </Button>
          <Button
            variant="ghost"
            style={{
              width: "100%",
              height: 30,
              justifyContent: "flex-start",
              gap: 8,
            }}
            onClick={() => pushPanel({ type: "person-notifications" })}
          >
            <span style={{ fontSize: 13, flex: 1 }}>Notifications</span>
            <span style={{ fontSize: 12, color: "var(--tt-gray-light-a-400)" }}>
              {config.notifications === "users-only"
                ? "Users only"
                : "Everyone"}
            </span>
            <ChevronRight
              style={{ width: 13, height: 13 }}
              className="tiptap-button-icon-sub"
            />
          </Button>
        </CardItemGroup>
      )}

      {/* Formula — drill into FormulaProperty */}
      {config.type === "formula" && (
        <CardItemGroup>
          <CardGroupLabel>Formula</CardGroupLabel>
          <Button
            variant="ghost"
            style={{
              width: "100%",
              height: 30,
              justifyContent: "flex-start",
              gap: 8,
            }}
            onClick={() => pushPanel({ type: "formula-editor" })}
          >
            <span
              style={{
                fontSize: 12,
                fontFamily: "var(--font-mono)",
                color: config.expression
                  ? "var(--tt-brand-color-400)"
                  : "var(--tt-gray-light-a-300)",
                flex: 1,
                overflow: "hidden",
                textOverflow: "ellipsis",
                whiteSpace: "nowrap",
                textAlign: "left",
              }}
            >
              {config.expression || "Add a formula"}
            </span>
            <ChevronRight
              style={{ width: 13, height: 13 }}
              className="tiptap-button-icon-sub"
            />
          </Button>
        </CardItemGroup>
      )}

      {/* ── Actions — hidden for title ── */}
      {!isTitle && (
        <CardItemGroup
          style={{
            borderTop: "0.5px solid var(--tt-border-color)",
            paddingTop: 4,
          }}
        >
          <FreezePropertyButton onFreeze={onFreeze} isFrozen={isFrozen} />
          <HidePropertyButton onHide={onHide} />
          <UnwrapPropertyButton onUnwrap={onUnwrap} isUnwrapped={isUnwrapped} />
          <Separator orientation="horizontal" />
          <DuplicatePropertyButton onDuplicate={onDuplicate} />
          <DeletePropertyButton onDelete={onDelete} />
        </CardItemGroup>
      )}
    </CardBody>
  );
}

// ─────────────────────────────────────────────────────────────────────────────
// PropertyEditPanel — orchestrates the navigation stack
// ─────────────────────────────────────────────────────────────────────────────

export function PropertyEditPanel({
  property,
  allProperties,
  onUpdate,
  onDelete,
  onDuplicate,
  /*onClose,*/
  panelStack,
  pushPanel,
  popPanel,
  isFrozen,
  /*isHidden,*/
  isUnwrapped,
  onFreeze,
  onHide,
  onUnwrap,
}: PropertyEditPanelProps) {
  const currentView = panelStack[panelStack.length - 1];
  const canGoBack = panelStack.length > 1;
  const { config } = property;
  const isTitle = config.type === "title";

  const updateConfig = (updated: PropertyConfig) =>
    onUpdate({ config: updated });

  const renderCurrentView = () => {
    switch (currentView.type) {
      case "main":
        return (
          <MainPanel
            property={property}
            allProperties={allProperties}
            onUpdate={onUpdate}
            onDelete={onDelete}
            onDuplicate={onDuplicate}
            pushPanel={pushPanel}
            isTitle={isTitle}
            onFreeze={onFreeze}
            isFrozen={isFrozen}
            onHide={onHide}
            onUnwrap={onUnwrap}
            isUnwrapped={isUnwrapped}
          />
        );

      case "status-editor":
        if (config.type !== "status") return null;
        return (
          <StatusProperty
            groups={config.groups}
            onChange={(groups) => updateConfig({ ...config, groups })}
          />
        );

      case "formula-editor":
        if (config.type !== "formula") return null;
        return (
          <FormulaProperty
            properties={allProperties.map((p) => ({
              id: p.id,
              name: p.name,
              type: p.config.type as any,
            }))}
            formula={config.expression}
            onChange={(expression) => updateConfig({ ...config, expression })}
            onClose={popPanel}
          />
        );

      case "person-limit":
      case "person-default":
      case "person-notifications":
        if (config.type !== "person") return null;
        return (
          <PersonProperty
            value={{
              limit: config.limit,
              default: config.default,
              notifications: config.notifications,
            }}
            onNavigate={(panel) =>
              pushPanel({ type: `person-${panel}` as PanelView["type"] })
            }
          />
        );

      default:
        return null;
    }
  };

  const headerTitle = (() => {
    switch (currentView.type) {
      case "main":
        return isTitle ? "Title property" : "Edit property";
      case "status-editor":
        return "Status options";
      case "formula-editor":
        return "Edit formula";
      case "person-limit":
        return "Limit";
      case "person-default":
        return "Default";
      case "person-notifications":
        return "Notifications";
      default:
        return "Edit property";
    }
  })();

  return (
    <Card>
      <PanelHeader
        title={headerTitle}
        canGoBack={canGoBack}
        onBack={popPanel}
      />
      {renderCurrentView()}
    </Card>
  );
}
