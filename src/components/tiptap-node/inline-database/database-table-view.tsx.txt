// ─────────────────────────────────────────────────────────────────────────────
// Cell renderers — read-only display per property type
// ─────────────────────────────────────────────────────────────────────────────
function CellValue({
  value,
  config,
}: {
  value: PropertyValue;
  config: PropertyConfig;
}) {
  if (value === null || value === undefined || value === "") {
    return null;
  }

  switch (config.type) {
    case "checkbox":
      return (
        <span
          style={{
            width: 14,
            height: 14,
            border: "1.5px solid",
            borderColor: value
              ? "var(--tt-brand-color-400)"
              : "var(--tt-gray-light-a-300)",
            borderRadius: 3,
            display: "flex",
            alignItems: "center",
            justifyContent: "center",
            background: value ? "var(--tt-brand-color-400)" : "transparent",
          }}
        >
          {value && (
            <svg width="9" height="9" viewBox="0 0 9 9" fill="none">
              <path
                d="M1.5 4.5l2.5 2.5 4-4"
                stroke="#fff"
                strokeWidth="1.5"
                strokeLinecap="round"
                strokeLinejoin="round"
              />
            </svg>
          )}
        </span>
      );

    case "select": {
      const opt = config.options.find(
        (o: SelectOption) => o.id === value || o.name === value,
      );
      if (!opt) return <span style={{ fontSize: 12 }}>{String(value)}</span>;
      return (
        <span
          style={{
            fontSize: 11,
            fontWeight: 500,
            padding: "1px 7px",
            borderRadius: 20,
            background: `color-mix(in srgb, ${dotColor(opt.color)} 15%, transparent)`,
            color: dotColor(opt.color),
          }}
        >
          {opt.name}
        </span>
      );
    }

    case "multi_select": {
      const ids = Array.isArray(value) ? (value as string[]) : [];
      return (
        <div style={{ display: "flex", flexWrap: "wrap", gap: 3 }}>
          {ids.map((id) => {
            const opt = config.options.find(
              (o: SelectOption) => o.id === id || o.name === id,
            );
            if (!opt) return null;
            return (
              <span
                key={id}
                style={{
                  fontSize: 11,
                  fontWeight: 500,
                  padding: "1px 7px",
                  borderRadius: 20,
                  background: `color-mix(in srgb, ${dotColor(opt.color)} 15%, transparent)`,
                  color: dotColor(opt.color),
                }}
              >
                {opt.name}
              </span>
            );
          })}
        </div>
      );
    }

    case "status": {
      const allItems = config.groups.flatMap((g: any) => g.items);
      const item = allItems.find(
        (i: any) => i.name === value || i.id === value,
      );
      if (!item) return <span style={{ fontSize: 12 }}>{String(value)}</span>;
      return (
        <span
          style={{
            display: "inline-flex",
            alignItems: "center",
            gap: 5,
            fontSize: 11,
            fontWeight: 500,
            padding: "1px 7px",
            borderRadius: 20,
            background: `color-mix(in srgb, ${statusDot(item.color)} 15%, transparent)`,
            color: statusDot(item.color),
          }}
        >
          <span
            style={{
              width: 6,
              height: 6,
              borderRadius: "50%",
              background: statusDot(item.color),
              flexShrink: 0,
            }}
          />
          {item.name}
        </span>
      );
    }

    case "date":
    case "created_time":
    case "edited_time": {
      const d = new Date(String(value));
      return <span style={{ fontSize: 12 }}>{d.toLocaleDateString()}</span>;
      // try {
      //

      // } catch {
      //   return <span style={{ fontSize: 12 }}>{String(value)}</span>;
      // }
    }

    case "person":
    case "created_by":
    case "edited_by": {
      const people = Array.isArray(value) ? value : [value];
      return (
        <div style={{ display: "flex", gap: 3 }}>
          {(people as any[]).map((p, i) => (
            <span
              key={i}
              style={{
                width: 20,
                height: 20,
                borderRadius: "50%",
                background: "var(--tt-brand-color-100)",
                color: "var(--tt-brand-color-600)",
                fontSize: 10,
                fontWeight: 500,
                display: "flex",
                alignItems: "center",
                justifyContent: "center",
              }}
            >
              {String(p?.name ?? p ?? "?")
                .charAt(0)
                .toUpperCase()}
            </span>
          ))}
        </div>
      );
    }

    case "number": {
      const n = Number(value);
      if (isNaN(n)) return null;
      const formatted = (() => {
        switch (config.format) {
          case "percent":
            return `${n}%`;
          case "dollar":
            return `$${n.toLocaleString()}`;
          case "euro":
            return `€${n.toLocaleString()}`;
          case "pound":
            return `£${n.toLocaleString()}`;
          case "yen":
            return `¥${n.toLocaleString()}`;
          case "number_with_commas":
            return n.toLocaleString();
          default:
            return String(n);
        }
      })();
      return <span style={{ fontSize: 13 }}>{formatted}</span>;
    }

    case "formula":
      return (
        <span
          style={{
            fontSize: 13,
            color: "var(--tt-gray-light-a-600)",
            fontStyle: "italic",
          }}
        >
          {String(value)}
        </span>
      );

    default:
      return <span style={{ fontSize: 13 }}>{String(value)}</span>;
  }
}

function dotColor(color: string): string {
  const MAP: Record<string, string> = {
    gray: "#9b9b9b",
    blue: "#3b82f6",
    green: "#22c55e",
    orange: "#f97316",
    red: "#ef4444",
    purple: "#8b5cf6",
    yellow: "#eab308",
    pink: "#ec4899",
    brown: "#8b6f5e",
  };
  return MAP[color] ?? "#9b9b9b";
}

function statusDot(color: string): string {
  return dotColor(color);
}

// ─────────────────────────────────────────────────────────────────────────────
// Inline cell editor
// ─────────────────────────────────────────────────────────────────────────────

interface CellEditorProps {
  value: PropertyValue;
  config: PropertyConfig;
  onSave: (value: PropertyValue) => void;
  onClose: () => void;
}

function CellEditor({ value, config, onSave, onClose }: CellEditorProps) {
  const [draft, setDraft] = useState(value);

  const commit = () => {
    onSave(draft);
    onClose();
  };

  const handleKey = (e: React.KeyboardEvent) => {
    if (e.key === "Enter" && !e.shiftKey) {
      e.preventDefault();
      commit();
    }
    if (e.key === "Escape") onClose();
  };

  switch (config.type) {
    case "checkbox":
      return (
        <button
          className="db-cell-editor db-cell-editor--checkbox"
          onClick={() => {
            onSave(!value);
            onClose();
          }}
        >
          <CellValue value={!value} config={config} />
        </button>
      );

    case "select":
    case "multi_select":
    case "status": {
      const allOpts =
        config.type === "status"
          ? config.groups.flatMap((g: any) => g.items)
          : config.options;

      return (
        <div className="db-cell-editor db-cell-editor--select">
          {allOpts.map((opt: any) => {
            const isSelected =
              config.type === "multi_select"
                ? Array.isArray(draft) &&
                  (draft as string[]).includes(opt.id ?? opt.name)
                : draft === opt.id || draft === opt.name;

            return (
              <button
                key={opt.id ?? opt.name}
                className={`db-select-option ${isSelected ? "db-select-option--selected" : ""}`}
                onClick={() => {
                  if (config.type === "multi_select") {
                    const current = Array.isArray(draft)
                      ? (draft as string[])
                      : [];
                    const id = opt.id ?? opt.name;
                    const next = current.includes(id)
                      ? current.filter((v) => v !== id)
                      : [...current, id];
                    setDraft(next);
                  } else {
                    onSave(opt.name);
                    onClose();
                  }
                }}
              >
                <CellValue value={opt.name} config={config} />
              </button>
            );
          })}
          {config.type === "multi_select" && (
            <button className="db-select-confirm" onClick={commit}>
              Done
            </button>
          )}
        </div>
      );
    }

    case "number":
      return (
        <input
          className="db-cell-editor db-cell-editor--input"
          type="number"
          autoFocus
          value={draft === null ? "" : String(draft)}
          onChange={(e) =>
            setDraft(e.target.value === "" ? null : Number(e.target.value))
          }
          onKeyDown={handleKey}
          onBlur={commit}
        />
      );

    case "date":
      return (
        <input
          className="db-cell-editor db-cell-editor--input"
          type="date"
          autoFocus
          value={draft ? String(draft).slice(0, 10) : ""}
          onChange={(e) => setDraft(e.target.value)}
          onKeyDown={handleKey}
          onBlur={commit}
        />
      );

    default:
      return (
        <input
          className="db-cell-editor db-cell-editor--input"
          autoFocus
          value={draft === null ? "" : String(draft ?? "")}
          onChange={(e) => setDraft(e.target.value)}
          onKeyDown={handleKey}
          onBlur={commit}
        />
      );
  }
}

// ─────────────────────────────────────────────────────────────────────────────
// Record row
// ─────────────────────────────────────────────────────────────────────────────
interface RecordRowProps {
  record: DatabaseRecord;
  properties: DatabaseProperty[];
  isEditing: (propertyId: string) => boolean;
  canEdit: (propertyId: string) => boolean;
  isFrozen: (propertyId: string) => boolean; // ← add
  isUnwrapped: (propertyId: string) => boolean; // ← add
  onCellClick: (propertyId: string) => void;
  onCellSave: (propertyId: string, value: PropertyValue) => void;
  onCellClose: () => void;
  onOpenRecord: () => void;
}

function RecordRow({
  record,
  properties,
  isEditing,
  canEdit,
  isFrozen,
  isUnwrapped,
  onCellClick,
  onCellSave,
  onCellClose,
  onOpenRecord,
}: RecordRowProps) {
  return (
    <tr className="db-tr">
      {properties.map((prop) => {
        const value = record.values[prop.id] ?? null;
        const editing = isEditing(prop.id);
        const editable = canEdit(prop.id);
        const isTitle = prop.config.type === "title";
        const frozen = isFrozen(prop.id);
        const unwrapped = isUnwrapped(prop.id);

        return (
          <td
            key={prop.id}
            className={`db-td
              ${isTitle ? "db-td--title" : ""}
              ${frozen ? "db-td--frozen" : ""}
              ${unwrapped ? "db-td--unwrapped" : ""}
            `}
            style={
              frozen ? { position: "sticky", left: 0, zIndex: 1 } : undefined
            }
            onClick={() => editable && onCellClick(prop.id)}
          >
            {isTitle && (
              <button
                className="db-open-record-btn"
                onClick={(e) => {
                  e.stopPropagation();
                  onOpenRecord();
                }}
                aria-label="Open record"
              >
                <Maximize2 style={{ width: 11, height: 11 }} />
              </button>
            )}

            {editing ? (
              <CellEditor
                value={value}
                config={prop.config}
                onSave={(v) => onCellSave(prop.id, v)}
                onClose={onCellClose}
              />
            ) : (
              <div className="db-cell-display">
                <CellValue value={value} config={prop.config} />
              </div>
            )}
          </td>
        );
      })}

      <td className="db-td db-td--empty" />
    </tr>
  );
}

// ─────────────────────────────────────────────────────────────────────────────
// Add property dropdown
// ─────────────────────────────────────────────────────────────────────────────

const QUICK_ADD_TYPES = [
  "text",
  "number",
  "select",
  "multi_select",
  "status",
  "checkbox",
  "date",
  "person",
] as const;

interface AddPropertyDropdownProps {
  onAdd: (type: any) => void;
  onClose: () => void;
}

function AddPropertyDropdown({ onAdd, onClose }: AddPropertyDropdownProps) {
  return (
    <div className="db-dropdown db-dropdown--property" onMouseLeave={onClose}>
      {QUICK_ADD_TYPES.map((type) => {
        const meta = getTypeMeta(type);
        const Icon = (LucideIcons as any)[meta.icon];
        return (
          <button
            key={type}
            className="db-dropdown-item"
            onClick={() => {
              onAdd(type);
              onClose();
            }}
          >
            {Icon && <Icon style={{ width: 13, height: 13 }} />}
            <span>{meta.label}</span>
          </button>
        );
      })}
    </div>
  );
}

// ─────────────────────────────────────────────────────────────────────────────
// DatabaseTableView
// ─────────────────────────────────────────────────────────────────────────────

interface DatabaseTableViewProps {
  db: UseDatabaseReturn;
}

export function DatabaseTableView({ db }: DatabaseTableViewProps) {
  const [openPropId, setOpenPropId] = useState<string | null>(null);
  const [addPropOpen, setAddPropOpen] = useState(false);
  const { isPropertyFrozen } = db;

  return (
    <div className="db-table-scroll">
      <table className="db-table" style={{ width: "100%" }}>
        <thead>
          <tr>
            {db.visibleProperties.map((prop) => {
              const meta = getTypeMeta(prop.config.type);
              const Icon = (LucideIcons as any)[meta.icon];
              const isOpen = openPropId === prop.id;
              const isFrozen = isPropertyFrozen(prop.id);
              return (
                <th
                  key={prop.id}
                  className={`db-th ${isOpen ? "db-th--active" : ""} ${isFrozen ? "db-th--frozen" : ""}`}
                  style={{
                    width: prop.width ?? 160,
                    minWidth: prop.width ?? 160,
                    padding: 0,
                    marginBottom: 0,
                    ...(isFrozen && {
                      position: "sticky",
                      left: 0,
                      zIndex: 2,
                    }),
                  }}
                >
                  <Popover
                    open={isOpen}
                    onOpenChange={(v) => setOpenPropId(v ? prop.id : null)}
                  >
                    <PopoverTrigger asChild>
                      <Button
                        variant="ghost"
                        style={{
                          width: "100%",
                          justifyContent: "flex-start",
                          borderRadius: "var(--tt-radius-sm)",
                        }}
                      >
                        {Icon && <Icon className="tiptap-button-icon" />}
                        <span className="tiptap-button-text">{prop.name}</span>
                      </Button>
                    </PopoverTrigger>
                    <PopoverContent side="right" align="center">
                      <PropertyEditPanel
                        property={prop}
                        allProperties={db.attrs.properties}
                        onUpdate={(patch) => db.updateProperty(prop.id, patch)}
                        onDelete={() => {
                          db.deleteProperty(prop.id);
                          setOpenPropId(null);
                        }}
                        onDuplicate={() => {
                          db.addProperty(
                            prop.config.type,
                            `${prop.name} (copy)`,
                          );
                          setOpenPropId(null);
                        }}
                        onClose={() => setOpenPropId(null)}
                        panelStack={db.panelStack}
                        pushPanel={db.pushPanel}
                        popPanel={db.popPanel}
                        resetPanel={db.resetPanel}
                        isFrozen={db.isPropertyFrozen(prop.id)}
                        isHidden={db.isPropertyHidden(prop.id)}
                        isUnwrapped={db.isPropertyUnwrapped(prop.id)}
                        onFreeze={() => db.freezeProperty(prop.id)}
                        onHide={() => db.hideProperty(prop.id)}
                        onUnwrap={() => db.unwrapProperty(prop.id)}
                      />
                    </PopoverContent>
                  </Popover>
                </th>
              );
            })}

            {/* Add property column */}
            <th className="db-th db-th--add">
              <ButtonGroup orientation="horizontal">
                <Popover open={addPropOpen} onOpenChange={setAddPropOpen}>
                  <PopoverTrigger asChild>
                    <Button variant="ghost" aria-label="Add property">
                      <Plus className="tiptap-button-icon" />
                    </Button>
                  </PopoverTrigger>
                  <PopoverContent side="right" align="start">
                    <AddPropertyDropdown
                      onAdd={(type) => {
                        db.addProperty(type);
                        setAddPropOpen(false);
                      }}
                      onClose={() => setAddPropOpen(false)}
                    />
                  </PopoverContent>
                </Popover>
                <Button variant="ghost">
                  <LucideIcons.Ellipsis className="tiptap-button-icon" />
                </Button>
              </ButtonGroup>
            </th>
          </tr>
        </thead>

        <tbody>
          {db.resolvedRecords.map((record) => (
            <RecordRow
              key={record.id}
              record={record}
              properties={db.visibleProperties}
              isEditing={(propId) => db.isCellEditing(record.id, propId)}
              canEdit={(propId) => db.canEditProperty(propId)}
              isFrozen={(propId) => db.isPropertyFrozen(propId)}
              isUnwrapped={(propId) => db.isPropertyUnwrapped(propId)}
              onCellClick={(propId) =>
                db.setEditingCell({ recordId: record.id, propertyId: propId })
              }
              onCellSave={(propId, value) =>
                db.updateRecord(record.id, { [propId]: value })
              }
              onCellClose={() => db.setEditingCell(null)}
              onOpenRecord={() => db.setOpenRecordId(record.id)}
            />
          ))}

          {/* New record row */}
          <tr className="db-tr db-tr--new">
            <td
              className={`db-td db-td--new `}
              colSpan={db.visibleProperties.length + 1}
            >
              <button
                className="db-new-record-btn"
                onClick={() => db.addRecord()}
              >
                <Plus style={{ width: 13, height: 13 }} />
                <span>New</span>
              </button>
            </td>
          </tr>
        </tbody>
      </table>
    </div>
  );
}
