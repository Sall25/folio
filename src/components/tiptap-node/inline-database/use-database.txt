/* eslint-disable @typescript-eslint/no-explicit-any */
import { useState, useMemo, useCallback } from "react";
import type { Editor } from "@tiptap/core";

import type {
  DatabaseAttrs,
  DatabaseProperty,
  DatabaseRecord,
  DatabaseView,
  DatabaseUIState,
  CellAddress,
  FilterGroup,
  SortRule,
  PropertyType,
  PropertyValue,
  PanelView,
  ID,
  TableView,
} from "./types";

import { isReadOnlyProperty, isTableView } from "./types";

// ─────────────────────────────────────────────────────────────────────────────
// Filter engine
// ─────────────────────────────────────────────────────────────────────────────

function applyFilters(
  records: DatabaseRecord[],
  filterGroups: FilterGroup[],
): DatabaseRecord[] {
  if (!filterGroups.length) return records;

  return records.filter((record) =>
    // All groups must pass (groups are AND-ed together)
    filterGroups.every((group) => {
      const results = group.rules.map((rule) => {
        const value = record.values[rule.propertyId];

        switch (rule.operator) {
          case "is_empty":
            return (
              value === null ||
              value === undefined ||
              value === "" ||
              (Array.isArray(value) && value.length === 0)
            );

          case "is_not_empty":
            return (
              value !== null &&
              value !== undefined &&
              value !== "" &&
              !(Array.isArray(value) && value.length === 0)
            );

          case "equals":
            return String(value) === String(rule.value);

          case "not_equals":
            return String(value) !== String(rule.value);

          case "contains":
            return String(value ?? "")
              .toLowerCase()
              .includes(String(rule.value ?? "").toLowerCase());

          case "not_contains":
            return !String(value ?? "")
              .toLowerCase()
              .includes(String(rule.value ?? "").toLowerCase());

          case "starts_with":
            return String(value ?? "")
              .toLowerCase()
              .startsWith(String(rule.value ?? "").toLowerCase());

          case "ends_with":
            return String(value ?? "")
              .toLowerCase()
              .endsWith(String(rule.value ?? "").toLowerCase());

          case "greater_than":
            return Number(value) > Number(rule.value);

          case "greater_than_or_equal":
            return Number(value) >= Number(rule.value);

          case "less_than":
            return Number(value) < Number(rule.value);

          case "less_than_or_equal":
            return Number(value) <= Number(rule.value);

          case "is_before":
            return new Date(String(value)) < new Date(String(rule.value));

          case "is_after":
            return new Date(String(value)) > new Date(String(rule.value));

          case "is_on_or_before":
            return new Date(String(value)) <= new Date(String(rule.value));

          case "is_on_or_after":
            return new Date(String(value)) >= new Date(String(rule.value));

          case "is_checked":
            return value === true;

          case "is_unchecked":
            return value === false;

          default:
            return true;
        }
      });

      // Rules within a group are AND-ed or OR-ed based on conjunction
      return group.conjunction === "and"
        ? results.every(Boolean)
        : results.some(Boolean);
    }),
  );
}

// ─────────────────────────────────────────────────────────────────────────────
// Sort engine
// ─────────────────────────────────────────────────────────────────────────────

function applySorts(
  records: DatabaseRecord[],
  sorts: SortRule[],
): DatabaseRecord[] {
  if (!sorts.length) return records;

  return [...records].sort((a, b) => {
    for (const sort of sorts) {
      const aVal = a.values[sort.propertyId];
      const bVal = b.values[sort.propertyId];

      let comparison = 0;

      if (aVal === null || aVal === undefined) comparison = 1;
      else if (bVal === null || bVal === undefined) comparison = -1;
      else if (typeof aVal === "boolean") {
        comparison = aVal === bVal ? 0 : aVal ? -1 : 1;
      } else if (typeof aVal === "number" && typeof bVal === "number") {
        comparison = aVal - bVal;
      } else {
        comparison = String(aVal).localeCompare(String(bVal));
      }

      if (comparison !== 0) {
        return sort.direction === "asc" ? comparison : -comparison;
      }
    }
    return 0;
  });
}

// ─────────────────────────────────────────────────────────────────────────────
// Formula evaluator — runs expression against a record's values
// ─────────────────────────────────────────────────────────────────────────────

function evaluateFormula(
  expression: string,
  record: DatabaseRecord,
  properties: DatabaseProperty[],
): PropertyValue {
  if (!expression.trim()) return null;

  try {
    // Build a context where each property name is a variable
    const ctx: Record<string, unknown> = {};
    for (const prop of properties) {
      const key = prop.name.replace(/\s+/g, "_");
      ctx[key] = record.values[prop.id] ?? null;
    }

    const keys = Object.keys(ctx);
    const vals = Object.values(ctx);
    // eslint-disable-next-line no-new-func
    const fn = new Function(...keys, `"use strict"; return (${expression});`);
    const result = fn(...vals);

    if (result instanceof Date) return result.toISOString();
    if (Array.isArray(result)) return result.join(", ");
    return String(result ?? "");
  } catch {
    return null;
  }
}

// ─────────────────────────────────────────────────────────────────────────────
// Hook
// ─────────────────────────────────────────────────────────────────────────────

export interface UseDatabaseReturn {
  // ── Derived from attrs ───────────────────────────────────────────────────
  attrs: DatabaseAttrs;
  activeView: DatabaseView | null;
  visibleProperties: DatabaseProperty[];
  filteredRecords: DatabaseRecord[];
  /** Records with formula values pre-computed */
  resolvedRecords: DatabaseRecord[];

  // ── Ephemeral UI state ───────────────────────────────────────────────────
  editingCell: CellAddress | null;
  openRecordId: ID | null;
  openPropertyId: ID | null;
  searchQuery: string;
  panelStack: PanelView[];

  // ── Record mutations ─────────────────────────────────────────────────────
  addRecord: () => void;
  updateRecord: (
    recordId: ID,
    values: Partial<DatabaseRecord["values"]>,
  ) => void;
  updateRecordBody: (recordId: ID, body: DatabaseRecord["body"]) => void;
  deleteRecord: (recordId: ID) => void;
  duplicateRecord: (recordId: ID) => void;

  // ── Property mutations ───────────────────────────────────────────────────
  addProperty: (type: PropertyType, name?: string) => void;
  updateProperty: (
    propertyId: ID,
    patch: Partial<Omit<DatabaseProperty, "id">>,
  ) => void;
  deleteProperty: (propertyId: ID) => void;
  reorderProperties: (orderedIds: ID[]) => void;

  // ── View mutations ───────────────────────────────────────────────────────
  setActiveView: (viewId: ID) => void;
  addView: (type: DatabaseView["type"], name: string) => void;
  updateView: (
    viewId: ID,
    patch: Partial<Omit<DatabaseView, "id" | "type">>,
  ) => void;
  deleteView: (viewId: ID) => void;

  // ── Ephemeral actions ────────────────────────────────────────────────────
  setEditingCell: (cell: CellAddress | null) => void;
  setOpenRecordId: (id: ID | null) => void;
  setOpenPropertyId: (id: ID | null) => void;
  setSearchQuery: (q: string) => void;

  // ── Panel navigation ─────────────────────────────────────────────────────
  pushPanel: (view: PanelView) => void;
  popPanel: () => void;
  resetPanel: () => void;
  currentPanel: PanelView;

  // ── Helpers ──────────────────────────────────────────────────────────────
  getRecord: (recordId: ID) => DatabaseRecord | undefined;
  getProperty: (propertyId: ID) => DatabaseProperty | undefined;
  isCellEditing: (recordId: ID, propertyId: ID) => boolean;
  canEditProperty: (propertyId: ID) => boolean;

  freezeProperty: (propertyId: ID) => void;
  hideProperty: (propertyId: ID) => void;
  unwrapProperty: (propertyId: ID) => void;
  isPropertyFrozen: (propertyId: ID) => boolean;
  isPropertyHidden: (propertyId: ID) => boolean;
  isPropertyUnwrapped: (propertyId: ID) => boolean;
}

export function useDatabase(
  attrs: DatabaseAttrs,
  editor: Editor,
): UseDatabaseReturn {
  const nodeId = attrs.id;

  // ── Ephemeral UI state ─────────────────────────────────────────────────────

  const [uiState, setUIState] = useState<DatabaseUIState>({
    editingCell: null,
    openRecordId: null,
    openPropertyId: null,
    searchQuery: "",
    panelStack: [{ type: "main" }],
  });

  const patchUI = useCallback(
    (patch: Partial<DatabaseUIState>) =>
      setUIState((prev) => ({ ...prev, ...patch })),
    [],
  );

  // ── Derived state ──────────────────────────────────────────────────────────

  const activeView = useMemo(
    () =>
      attrs.views.find((v) => v.id === attrs.activeViewId) ??
      attrs.views[0] ??
      null,
    [attrs.views, attrs.activeViewId],
  );

  const visibleProperties = useMemo(() => {
    if (!activeView) return attrs.properties;
    const hidden = new Set(activeView.hiddenProperties);

    if (isTableView(activeView)) {
      // Respect the explicit propertyOrder for table views
      const orderMap = new Map(
        (activeView as TableView).propertyOrder.map((id, i) => [id, i]),
      );
      return attrs.properties
        .filter((p) => !hidden.has(p.id))
        .sort(
          (a, b) => (orderMap.get(a.id) ?? 999) - (orderMap.get(b.id) ?? 999),
        );
    }

    return attrs.properties.filter((p) => !hidden.has(p.id));
  }, [attrs.properties, activeView]);

  const filteredRecords = useMemo(() => {
    if (!activeView) return attrs.records;

    let result = attrs.records;

    // Search — matches against the title property
    if (uiState.searchQuery.trim()) {
      const q = uiState.searchQuery.toLowerCase();
      const titleProp = attrs.properties.find((p) => p.config.type === "title");
      if (titleProp) {
        result = result.filter((r) =>
          String(r.values[titleProp.id] ?? "")
            .toLowerCase()
            .includes(q),
        );
      }
    }

    result = applyFilters(result, activeView.filters);
    result = applySorts(result, activeView.sorts);

    return result;
  }, [attrs.records, attrs.properties, activeView, uiState.searchQuery]);

  // Resolve formula values for every filtered record
  const resolvedRecords = useMemo(() => {
    const formulaProps = attrs.properties.filter(
      (p) => p.config.type === "formula",
    );
    if (!formulaProps.length) return filteredRecords;

    return filteredRecords.map((record) => {
      const computed: DatabaseRecord["values"] = { ...record.values };
      for (const prop of formulaProps) {
        if (prop.config.type !== "formula") continue;
        computed[prop.id] = evaluateFormula(
          prop.config.expression,
          record,
          attrs.properties,
        );
      }
      return { ...record, values: computed };
    });
  }, [filteredRecords, attrs.properties]);

  // ── Record mutations ───────────────────────────────────────────────────────

  const addRecord = useCallback(() => {
    editor.commands.addDatabaseRecord(nodeId);
  }, [editor, nodeId]);

  const updateRecord = useCallback(
    (recordId: ID, values: Partial<DatabaseRecord["values"]>) => {
      editor.commands.updateDatabaseRecord(nodeId, recordId, values);
    },
    [editor, nodeId],
  );

  const updateRecordBody = useCallback(
    (recordId: ID, body: DatabaseRecord["body"]) => {
      editor.commands.updateDatabaseRecord(nodeId, recordId, {});
      editor.commands.updateDatabaseRecord(nodeId, recordId, body as any);
    },
    [editor, nodeId],
  );

  const deleteRecord = useCallback(
    (recordId: ID) => {
      editor.commands.deleteDatabaseRecord(nodeId, recordId);
      if (uiState.openRecordId === recordId) {
        patchUI({ openRecordId: null });
      }
    },
    [editor, nodeId, uiState.openRecordId, patchUI],
  );

  const duplicateRecord = useCallback(
    (recordId: ID) => {
      const record = attrs.records.find((r) => r.id === recordId);
      if (!record) return;

      editor.commands.addDatabaseRecord(nodeId);
      // After adding, find the new record (last one) and copy values over
      const current = editor.getJSON();
      const dbNode = current.content?.find(
        (n: any) => n.type === "database" && n.attrs?.id === nodeId,
      );
      if (!dbNode?.attrs?.records) return;
      const newRecord: DatabaseRecord =
        dbNode.attrs.records[dbNode.attrs.records.length - 1];
      if (!newRecord) return;

      editor.commands.updateDatabaseRecord(nodeId, newRecord.id, record.values);
    },
    [editor, nodeId, attrs.records],
  );

  // ── Property mutations ─────────────────────────────────────────────────────

  const addProperty = useCallback(
    (type: PropertyType, name?: string) => {
      editor.commands.addDatabaseProperty(nodeId, type, name);
    },
    [editor, nodeId],
  );

  const updateProperty = useCallback(
    (propertyId: ID, patch: Partial<Omit<DatabaseProperty, "id">>) => {
      editor.commands.updateDatabaseProperty(nodeId, propertyId, patch);
    },
    [editor, nodeId],
  );

  const deleteProperty = useCallback(
    (propertyId: ID) => {
      editor.commands.deleteDatabaseProperty(nodeId, propertyId);
      if (uiState.openPropertyId === propertyId) {
        patchUI({ openPropertyId: null });
      }
    },
    [editor, nodeId, uiState.openPropertyId, patchUI],
  );

  const reorderProperties = useCallback(
    (orderedIds: ID[]) => {
      editor.commands.reorderDatabaseProperties(nodeId, orderedIds);
    },
    [editor, nodeId],
  );

  // ── View mutations ─────────────────────────────────────────────────────────

  const setActiveView = useCallback(
    (viewId: ID) => {
      editor.commands.setDatabaseView(nodeId, viewId);
      // Reset ephemeral state when switching views
      patchUI({ editingCell: null, searchQuery: "" });
    },
    [editor, nodeId, patchUI],
  );

  const addView = useCallback(
    (type: DatabaseView["type"], name: string) => {
      editor.commands.addDatabaseView(nodeId, type, name);
    },
    [editor, nodeId],
  );

  const updateView = useCallback(
    (viewId: ID, patch: Partial<Omit<DatabaseView, "id" | "type">>) => {
      editor.commands.updateDatabaseView(nodeId, viewId, patch);
    },
    [editor, nodeId],
  );

  const deleteView = useCallback(
    (viewId: ID) => {
      editor.commands.deleteDatabaseView(nodeId, viewId);
    },
    [editor, nodeId],
  );

  // ── Ephemeral actions ──────────────────────────────────────────────────────

  const setEditingCell = useCallback(
    (cell: CellAddress | null) => patchUI({ editingCell: cell }),
    [patchUI],
  );

  const setOpenRecordId = useCallback(
    (id: ID | null) => patchUI({ openRecordId: id, editingCell: null }),
    [patchUI],
  );

  const setOpenPropertyId = useCallback(
    (id: ID | null) =>
      patchUI({
        openPropertyId: id,
        panelStack: [{ type: "main" }], // always reset to main when opening a new property
      }),
    [patchUI],
  );

  const setSearchQuery = useCallback(
    (q: string) => patchUI({ searchQuery: q }),
    [patchUI],
  );

  // ── Panel navigation ───────────────────────────────────────────────────────

  const pushPanel = useCallback(
    (view: PanelView) =>
      setUIState((prev) => ({
        ...prev,
        panelStack: [...prev.panelStack, view],
      })),
    [],
  );

  const popPanel = useCallback(
    () =>
      setUIState((prev) => ({
        ...prev,
        panelStack:
          prev.panelStack.length > 1
            ? prev.panelStack.slice(0, -1)
            : prev.panelStack,
      })),
    [],
  );

  const resetPanel = useCallback(
    () => patchUI({ panelStack: [{ type: "main" }] }),
    [patchUI],
  );

  // ── Helpers ────────────────────────────────────────────────────────────────

  const getRecord = useCallback(
    (recordId: ID) => attrs.records.find((r) => r.id === recordId),
    [attrs.records],
  );

  const getProperty = useCallback(
    (propertyId: ID) => attrs.properties.find((p) => p.id === propertyId),
    [attrs.properties],
  );

  const isCellEditing = useCallback(
    (recordId: ID, propertyId: ID) =>
      uiState.editingCell?.recordId === recordId &&
      uiState.editingCell?.propertyId === propertyId,
    [uiState.editingCell],
  );

  const canEditProperty = useCallback(
    (propertyId: ID) => {
      const prop = attrs.properties.find((p) => p.id === propertyId);
      if (!prop) return false;
      return !isReadOnlyProperty(prop.config.type);
    },
    [attrs.properties],
  );

  const freezeProperty = useCallback(
    (propertyId: ID) =>
      editor.commands.freezeDatabaseProperty(nodeId, propertyId),
    [editor, nodeId],
  );

  const hideProperty = useCallback(
    (propertyId: ID) =>
      editor.commands.hideDatabaseProperty(
        nodeId,
        propertyId,
        attrs.activeViewId,
      ),
    [editor, nodeId, attrs.activeViewId],
  );

  const unwrapProperty = useCallback(
    (propertyId: ID) =>
      editor.commands.unwrapDatabaseProperty(
        nodeId,
        propertyId,
        attrs.activeViewId,
      ),
    [editor, nodeId, attrs.activeViewId],
  );

  const isPropertyFrozen = useCallback(
    (propertyId: ID) => {
      const view = activeView as TableView | null;
      return view?.frozenPropertyId === propertyId;
    },
    [activeView],
  );

  const isPropertyHidden = useCallback(
    (propertyId: ID) =>
      activeView?.hiddenProperties.includes(propertyId) ?? false,
    [activeView],
  );

  const isPropertyUnwrapped = useCallback(
    (propertyId: ID) => {
      const view = activeView as TableView | null;
      return view?.unwrappedProperties?.includes(propertyId) ?? false;
    },
    [activeView],
  );

  return {
    // Derived
    attrs,
    activeView,
    visibleProperties,
    filteredRecords,
    resolvedRecords,

    // Ephemeral
    editingCell: uiState.editingCell,
    openRecordId: uiState.openRecordId,
    openPropertyId: uiState.openPropertyId,
    searchQuery: uiState.searchQuery,
    panelStack: uiState.panelStack,
    currentPanel: uiState.panelStack[uiState.panelStack.length - 1],

    // Record mutations
    addRecord,
    updateRecord,
    updateRecordBody,
    deleteRecord,
    duplicateRecord,

    // Property mutations
    addProperty,
    updateProperty,
    deleteProperty,
    reorderProperties,

    // View mutations
    setActiveView,
    addView,
    updateView,
    deleteView,

    // Ephemeral actions
    setEditingCell,
    setOpenRecordId,
    setOpenPropertyId,
    setSearchQuery,

    // Panel navigation
    pushPanel,
    popPanel,
    resetPanel,

    // Helpers
    getRecord,
    getProperty,
    isCellEditing,
    canEditProperty,
    freezeProperty,
    hideProperty,
    unwrapProperty,
    isPropertyFrozen,
    isPropertyHidden,
    isPropertyUnwrapped,
  };
}
